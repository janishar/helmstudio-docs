import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

## Four rooms, one gallery between them

<CardGrid>
  <Card title="Create" icon="pencil">
    Pick a model, write a prompt, make takes. Call your characters and places by name with @, and set a look once.
  </Card>
  <Card title="Gallery" icon="seti:image">
    Every take lands here with the settings and inputs that made it. Send any take to Create, a pipeline or the
    timeline.
  </Card>
  <Card title="Pipelines" icon="random">
    Chain workflows and run them over a list while you are away. Agent nodes judge, rewrite and retry.
  </Card>
  <Card title="Timeline" icon="seti:video">
    Cut takes into a sequence, direct new shots and voice lines in place, and export an MP4.
  </Card>
</CardGrid>

## Runs where your GPUs are

<CardGrid>
  <Card title="Apple Silicon" icon="laptop">
    MLX, PyTorch MPS and native runners share one memory budget on each Mac.
  </Card>
  <Card title="NVIDIA hosts" icon="rocket">
    Add CUDA machines. Each run goes to a host with the weights and the memory, or to the one you pick.
  </Card>
  <Card title="Private per person" icon="seti:lock">
    Every user sees only their own takes, sessions and pipelines. Accounts come from the admin.
  </Card>
  <Card title="Agents welcome" icon="puzzle">
    An in-app Assistant works every screen, and `helm mcp` gives Claude and other agents the same tools.
  </Card>
</CardGrid>
